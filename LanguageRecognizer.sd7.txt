$ include "seed7_05.s7i";

const type: StringArray is array string;

var StringArray: tokenList is 0 times "";
var integer: currentToken is 1;
var string: errorMessage is "";

var StringArray: shapeList is 0 times "";
var StringArray: coordOneList is 0 times "";
var StringArray: coordTwoList is 0 times "";
var StringArray: coordThreeList is 0 times "";

var integer: instructionCount is 0;
var boolean: running is TRUE;


# ============================================================
# DISPLAY GRAMMAR
# ============================================================

const proc: displayGrammar is func
begin
  writeln;
  writeln("==================================================");
  writeln("              LANGUAGE RECOGNIZER");
  writeln("==================================================");
  writeln;
  writeln("BNF GRAMMAR");
  writeln;
  writeln("<program>      -> begin <instructions> end");
  writeln("<instructions> -> <instruction>");
  writeln("                | <instruction> . <instructions>");
  writeln("<instruction>  -> SQR <coord>-<coord>");
  writeln("                | TRI <coord>-<coord>-<coord>");
  writeln("<coord>        -> <x><y>");
  writeln("<x>            -> A | B | C | D | E | F | G");
  writeln("<y>            -> 1 | 2 | 3 | 4 | 5 | 6");
  writeln;
  writeln("Example:");
  writeln("begin SQR A1-C4. TRI A1-C4-G3 end");
  writeln;
  writeln("Type EXIT to quit.");
  writeln("==================================================");
  writeln;
end func;


# ============================================================
# TOKENIZER
# ============================================================

const proc: tokenize (in string: inputString) is func

local
  var string: currentText is "";
  var char: currentChar is ' ';
  var integer: i is 0;

begin

  tokenList := 0 times "";
  currentToken := 1;
  currentText := "";

  if length(inputString) > 0 then

    for i range 1 to length(inputString) do

      currentChar := inputString[i];

      if currentChar = ' ' then

        if currentText <> "" then
          tokenList &:= currentText;
          currentText := "";
        end if;

      elsif currentChar = '.' then

        if currentText <> "" then
          tokenList &:= currentText;
          currentText := "";
        end if;

        tokenList &:= ".";

      elsif currentChar = '-' then

        if currentText <> "" then
          tokenList &:= currentText;
          currentText := "";
        end if;

        tokenList &:= "-";

      else

        currentText &:= currentChar;

      end if;

    end for;

  end if;

  if currentText <> "" then
    tokenList &:= currentText;
  end if;

end func;


# ============================================================
# GET CURRENT TOKEN
# ============================================================

const func string: getToken is func

result
  var string: tokenValue is "";

begin

  if currentToken <= length(tokenList) then
    tokenValue := tokenList[currentToken];
  end if;

end func;


# ============================================================
# MOVE TO NEXT TOKEN
# ============================================================

const proc: nextToken is func
begin

  if currentToken < length(tokenList) then
    currentToken +:= 1;
  else
    currentToken := length(tokenList) + 1;
  end if;

end func;


# ============================================================
# VALIDATE X
# ============================================================

const func boolean: validX (in char: value) is func

result
  var boolean: valid is FALSE;

begin

  if value = 'A' then
    valid := TRUE;
  elsif value = 'B' then
    valid := TRUE;
  elsif value = 'C' then
    valid := TRUE;
  elsif value = 'D' then
    valid := TRUE;
  elsif value = 'E' then
    valid := TRUE;
  elsif value = 'F' then
    valid := TRUE;
  elsif value = 'G' then
    valid := TRUE;
  end if;

end func;


# ============================================================
# VALIDATE Y
# ============================================================

const func boolean: validY (in char: value) is func

result
  var boolean: valid is FALSE;

begin

  if value = '1' then
    valid := TRUE;
  elsif value = '2' then
    valid := TRUE;
  elsif value = '3' then
    valid := TRUE;
  elsif value = '4' then
    valid := TRUE;
  elsif value = '5' then
    valid := TRUE;
  elsif value = '6' then
    valid := TRUE;
  end if;

end func;


# ============================================================
# VALIDATE COORDINATE
# ============================================================

const func boolean: validCoord (in string: coordinate) is func

result
  var boolean: valid is FALSE;

begin

  if length(coordinate) <> 2 then

    errorMessage :=
      "Error: Coordinate must contain one letter and one number.";

  elsif validX(coordinate[1]) = FALSE then

    errorMessage :=
      "Error: Coordinate letter must be A through G.";

  elsif validY(coordinate[2]) = FALSE then

    errorMessage :=
      "Error: Coordinate number must be 1 through 6.";

  else

    valid := TRUE;

  end if;

end func;


# ============================================================
# PARSE SQR
# ============================================================

const func boolean: parseSqr is func

result
  var boolean: valid is FALSE;

local
  var string: firstCoord is "";
  var string: secondCoord is "";

begin

  # Current token is SQR.

  nextToken;
  firstCoord := getToken;

  if validCoord(firstCoord) = TRUE then

    nextToken;

    if getToken = "-" then

      nextToken;
      secondCoord := getToken;

      if validCoord(secondCoord) = TRUE then

        shapeList &:= "SQR";
        coordOneList &:= firstCoord;
        coordTwoList &:= secondCoord;
        coordThreeList &:= "";

        instructionCount +:= 1;

        valid := TRUE;

      end if;

    else

      errorMessage :=
        "Error: SQR requires '-' between coordinates.";

    end if;

  end if;

end func;


# ============================================================
# PARSE TRI
# ============================================================

const func boolean: parseTri is func

result
  var boolean: valid is FALSE;

local
  var string: firstCoord is "";
  var string: secondCoord is "";
  var string: thirdCoord is "";

begin

  # Current token is TRI.

  nextToken;
  firstCoord := getToken;

  if validCoord(firstCoord) = TRUE then

    nextToken;

    if getToken = "-" then

      nextToken;
      secondCoord := getToken;

      if validCoord(secondCoord) = TRUE then

        nextToken;

        if getToken = "-" then

          nextToken;
          thirdCoord := getToken;

          if validCoord(thirdCoord) = TRUE then

            shapeList &:= "TRI";
            coordOneList &:= firstCoord;
            coordTwoList &:= secondCoord;
            coordThreeList &:= thirdCoord;

            instructionCount +:= 1;

            valid := TRUE;

          end if;

        else

          errorMessage :=
            "Error: TRI requires three coordinates.";

        end if;

      end if;

    else

      errorMessage :=
        "Error: TRI requires '-' between coordinates.";

    end if;

  end if;

end func;


# ============================================================
# PARSE INSTRUCTION
# ============================================================

const func boolean: parseInstruction is func

result
  var boolean: valid is FALSE;

local
  var string: instruction is "";

begin

  instruction := getToken;

  if instruction = "SQR" then

    valid := parseSqr;

  elsif instruction = "TRI" then

    valid := parseTri;

  else

    errorMessage :=
      "Error: Expected SQR or TRI.";

  end if;

end func;


# ============================================================
# PARSE INSTRUCTIONS
# ============================================================

const func boolean: parseInstructions is func

result
  var boolean: valid is FALSE;

local
  var string: tokenValue is "";

begin

  if parseInstruction = TRUE then

    valid := TRUE;

    nextToken;
    tokenValue := getToken;

    while tokenValue = "." do

      nextToken;

      if parseInstruction = TRUE then

        nextToken;
        tokenValue := getToken;

      else

        valid := FALSE;
        tokenValue := "";

      end if;

    end while;

  end if;

end func;


# ============================================================
# PARSE COMPLETE PROGRAM
# ============================================================

const func boolean: parseProgram (in string: inputString) is func

result
  var boolean: valid is FALSE;

local
  var string: tokenValue is "";

begin

  errorMessage := "";

  shapeList := 0 times "";
  coordOneList := 0 times "";
  coordTwoList := 0 times "";
  coordThreeList := 0 times "";

  instructionCount := 0;

  tokenize(inputString);

  if length(tokenList) = 0 then

    errorMessage :=
      "Error: No input was entered.";

  else

    tokenValue := getToken;

    # Check BEGIN.

    if tokenValue <> "begin" then

      errorMessage :=
        "Error: Program must begin with 'begin'.";

    else

      nextToken;

      # Check instructions.

      if parseInstructions = TRUE then

        tokenValue := getToken;

        # Check END.

        if tokenValue = "end" then

          nextToken;

          # Nothing may follow END.

          if currentToken > length(tokenList) then

            valid := TRUE;

          else

            errorMessage :=
              "Error: Unexpected text after 'end'.";

          end if;

        else

          errorMessage :=
            "Error: Program must end with 'end'.";

        end if;

      end if;

    end if;

  end if;

end func;


# ============================================================
# CREATE INSTRUCTION TEXT
# ============================================================

const func string: instructionText (in integer: position) is func

result
  var string: text is "";

begin

  if shapeList[position] = "SQR" then

    text :=
      "SQR " &
      coordOneList[position] &
      "-" &
      coordTwoList[position];

  else

    text :=
      "TRI " &
      coordOneList[position] &
      "-" &
      coordTwoList[position] &
      "-" &
      coordThreeList[position];

  end if;

end func;


# ============================================================
# DISPLAY RECOGNIZED INSTRUCTIONS
# ============================================================

const proc: displayInstructions is func

local
  var integer: i is 0;

begin

  writeln;
  writeln("Recognized instructions:");

  for i range 1 to instructionCount do

    write("  ");
    write(i);
    write(". ");
    writeln(instructionText(i));

  end for;

  writeln;

end func;


# ============================================================
# DISPLAY RIGHTMOST DERIVATION
# ============================================================

const proc: displayDerivation is func

local
  var integer: i is 0;

begin

  writeln;
  writeln("==================================================");
  writeln("              RIGHTMOST DERIVATION");
  writeln("==================================================");

  writeln("1. begin <instructions> end");

  if instructionCount = 1 then

    writeln("2. begin <instruction> end");

  else

    writeln("2. begin <instruction> . <instructions> end");

  end if;

  for i range 1 to instructionCount do

    if shapeList[i] = "SQR" then

      write("   <instruction> => SQR ");
      write(coordOneList[i]);
      write("-");
      writeln(coordTwoList[i]);

    else

      write("   <instruction> => TRI ");
      write(coordOneList[i]);
      write("-");
      write(coordTwoList[i]);
      write("-");
      writeln(coordThreeList[i]);

    end if;

  end for;

  writeln;
  writeln("Final sentence:");

  write("begin ");

  for i range 1 to instructionCount do

    write(instructionText(i));

    if i < instructionCount then
      write(". ");
    end if;

  end for;

  writeln(" end");
  writeln;

end func;


# ============================================================
# DISPLAY PARSE TREE
# ============================================================

const proc: displayParseTree is func

local
  var integer: i is 0;

begin

  writeln;
  writeln("==================================================");
  writeln("                   PARSE TREE");
  writeln("==================================================");

  writeln("                    <program>");
  writeln("                   /    |    \\");
  writeln("                begin  <instructions>  end");
  writeln("                         |");

  for i range 1 to instructionCount do

    if shapeList[i] = "SQR" then

      writeln("                    <instruction>");
      writeln("                     /    |    \\");
      writeln("                   SQR  <coord> <coord>");

      write("                        ");
      write(coordOneList[i]);
      write("     ");
      writeln(coordTwoList[i]);

    else

      writeln("                    <instruction>");
      writeln("                    /    |    |    \\");
      writeln("                  TRI  <coord> <coord> <coord>");

      write("                       ");
      write(coordOneList[i]);
      write("     ");
      write(coordTwoList[i]);
      write("     ");
      writeln(coordThreeList[i]);

    end if;

    if i < instructionCount then
      writeln("                         .");
    end if;

  end for;

  writeln;

end func;


# ============================================================
# DISPLAY SUCCESS
# ============================================================

const proc: displaySuccess is func
begin

  writeln;
  writeln("==================================================");
  writeln("                 VALID PROGRAM");
  writeln("==================================================");
  writeln("The input matches the grammar.");
  writeln;

end func;


# ============================================================
# DISPLAY ERROR
# ============================================================

const proc: displayError is func
begin

  writeln;
  writeln("==================================================");
  writeln("                INVALID PROGRAM");
  writeln("==================================================");

  if errorMessage <> "" then
    writeln(errorMessage);
  else
    writeln("Error: The input does not match the grammar.");
  end if;

  writeln;

end func;


# ============================================================
# MAIN
# ============================================================

const proc: main is func

local
  var string: inputString is "";
  var boolean: recognized is FALSE;

begin

  while running do

    displayGrammar;

    write("Enter your program: ");
    readln(inputString);

    if inputString = "EXIT" then

      running := FALSE;

    else

      recognized := parseProgram(inputString);

      if recognized = TRUE then

        displaySuccess;
        displayInstructions;
        displayDerivation;
        displayParseTree;

      else

        displayError;

      end if;

      writeln("Press ENTER to continue.");
      readln(inputString);

    end if;

  end while;

  writeln;
  writeln("Program terminated.");

end func;
